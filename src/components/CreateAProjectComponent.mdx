import {ADRI_URLS} from '@site/src/constants/urls';

1. Open <a href={ADRI_URLS.projects}>getadri.ai/projects</a> after logging in.
2. On the **Projects** page, click **Create Project**.
3. Enter the project details.
4. Click **Create** to confirm.

**Done when:** You are redirected to your newly created project.
